async function openDrawer(name) { const drawer = document.getElementById('drawer'); const drawerTitle = document.getElementById('drawer-title'); const drawerContent = document.getElementById('drawer-content'); drawerTitle.textContent = name; drawerContent.innerHTML = '
Loading cards...
'; drawer.classList.add('open'); try { const response = await fetch('/.netlify/functions/getBoard?group=' + encodeURIComponent(name) + '&_t=' + Date.now()); const data = await response.json(); if (data.items && data.items.length > 0) { drawerContent.innerHTML = data.items.map(item => `

${item.name}

${item.state || ''}

`).join(''); } else { drawerContent.innerHTML = '

No cards found in this group.

'; } } catch (error) { drawerContent.innerHTML = '

Error loading cards. Please try again.

'; } }
