function handleVoiceIntake(text) { showStatus("⚡ FRESH: Sissy is organizing your note..."); // 1. CLIENT-SIDE VAULT RADAR (BEFORE THE HOOK CALL) let detectedClient = null; const cleanSpoken = text.toLowerCase().replace(/[^a-z0-9\s]/g, ' '); const spokenWords = cleanSpoken.split(/\s+/).filter(w => w.length > 2); if (vaultData && vaultData.length > 0) { for (let row of vaultData) { const rawName = (row['Name'] || `${row['First name'] || ''} ${row['Last name'] || ''}`).trim(); const lowerName = rawName.toLowerCase(); const parts = lowerName.split(/\s+/).filter(p => p.length > 2); // Exact match or full name match if (cleanSpoken.includes(lowerName)) { detectedClient = row; break; } // Match both first & last name parts if spoken together if (parts.length >= 2 && parts.every(part => cleanSpoken.includes(part))) { detectedClient = row; break; } } } // 2. PROTECTIVE FALLBACK: NEVER DEFAULT TO SHOP NAME ON CUSTOMER INTAKE let initialName = detectedClient ? (detectedClient['Name'] || "Valued Client") : ""; if (!initialName && !cleanSpoken.includes("shop note") && !cleanSpoken.includes("internal")) { // Extract first two capitalized words or use first 3 spoken words as name placeholder const words = text.trim().split(/\s+/); initialName = words.slice(0, 2).join(' '); } // Display indicator const threadBox = document.getElementById('conversation-thread'); const bubbles = document.getElementById('thread-bubbles'); if (threadBox) threadBox.classList.remove('hidden'); if (bubbles) { bubbles.innerHTML += `
Fresh is typing... ⏳
`; bubbles.scrollTop = bubbles.scrollHeight; } // 3. SEND TO ENGINE WITH PRE-IDENTIFIED CLIENT fetch(SISSY_HOOK, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ transcript: text, clientName: initialName, timestamp: new Date().toISOString() }) }) .then(async res => cleanAndParseJSON(await res.text())) .then(data => { let clientName = data.clientName || initialName || MASTER_CARD.clientName; let finalAddress = detectedClient ? formatRowAddress(detectedClient) : MASTER_CARD.address; let finalPhone = detectedClient ? formatRowPhone(detectedClient) : MASTER_CARD.phone; const isScheduled = data.isScheduled === true || text.toLowerCase().includes('tomorrow') || text.toLowerCase().includes(' am') || text.toLowerCase().includes(' pm'); const card = { id: Date.now(), text: data.notes || text, clientName: clientName, address: finalAddress, phone: finalPhone, displayDate: data.displayDate || (text.toLowerCase().includes('tomorrow') ? 'Tomorrow' : 'Today'), time: data.time || '11:00 AM', timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }; const store = getStorage(); if (isScheduled) { store.scheduledCards.unshift(card); showStatus(`⚡ Scheduled: ${clientName}`); } else { store.pinballQueue.unshift(card); showStatus(`⚡ Queued: ${clientName}`); } saveStorage(store); render(); updateSissyConsole(`Locked in appointment for ${clientName}!`, text); }) .catch(() => { // Safe local offline dock const store = getStorage(); store.scheduledCards.unshift({ id: Date.now(), text: text, clientName: initialName, address: detectedClient ? formatRowAddress(detectedClient) : MASTER_CARD.address, phone: detectedClient ? formatRowPhone(detectedClient) : MASTER_CARD.phone, displayDate: 'Tomorrow', time: '11:00 AM', timestamp: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }) }); saveStorage(store); render(); showStatus(`⚡ Saved locally: ${initialName}`); }); }
