// ========================================== // PINBALL ACTION ZONE: FETCH LIVE INBOUND CARDS // ========================================== async function fetchInboundCards() { const syncBtn = document.getElementById('syncCardsBtn'); if (syncBtn) { syncBtn.disabled = true; syncBtn.innerText = '🔄 Syncing...'; } const FETCH_URL = 'https://hook.us2.make.com/5q1gs84brtie39vq4it42ex92of3959r'; try { const response = await fetch(FETCH_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ action: 'fetch_pinball_cards' }) }); if (!response.ok) { throw new Error(`Fetch error: ${response.status}`); } const data = await response.json(); const items = data.items || []; renderPinballCards(items); if (syncBtn) { syncBtn.innerText = `✅ Synced (${items.length})`; setTimeout(() => { syncBtn.innerText = '🔄 Sync Cards'; syncBtn.disabled = false; }, 3000); } } catch (err) { console.error('Failed to fetch cards:', err); if (syncBtn) { syncBtn.innerText = '⚠️ Sync Failed'; setTimeout(() => { syncBtn.innerText = '🔄 Sync Cards'; syncBtn.disabled = false; }, 3000); } } }
