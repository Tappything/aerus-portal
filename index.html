// ========================================== // TAPPYTHING / FRESH FOREVER CARD ENGINE // ========================================== // 1. OPEN WORK CARD DRAWER WITH FULL VAULT DATA function openWorkCard(clientId) { const drawer = document.getElementById('forever-work-drawer') || document.getElementById('activeWorkspaceDrawer'); if (drawer) { drawer.classList.remove('hidden'); drawer.classList.add('active'); // Lock active state to prevent data loss on mobile browser refresh localStorage.setItem('active_card_id', clientId); // Auto-populate customer data if available if (typeof openCustomerForeverCard === 'function') { openCustomerForeverCard(clientId); } } } // 2. CLOSE WORK CARD DRAWER & CLEAR CACHED STATE function closeWorkCard() { const drawer = document.getElementById('forever-work-drawer') || document.getElementById('activeWorkspaceDrawer'); if (drawer) { drawer.classList.add('hidden'); drawer.classList.remove('active'); } // Clear cached ID so it doesn't reopen unexpectedly localStorage.removeItem('active_card_id'); } // 3. AUTO-RESTORE ACTIVE CARD ON PAGE REFRESH (ZERO DATA LOSS) window.addEventListener('DOMContentLoaded', () => { const savedId = localStorage.getItem('active_card_id'); if (savedId) { openWorkCard(savedId); } });
